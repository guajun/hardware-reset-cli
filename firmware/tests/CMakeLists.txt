cmake_minimum_required(VERSION 3.22)
project(hwreset_native C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_library(reset_protocol OBJECT ../Core/Src/reset_protocol.c)
target_include_directories(reset_protocol PUBLIC ../Core/Inc)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(reset_protocol PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(reset_protocol_tests test_protocol.c)
target_link_libraries(reset_protocol_tests PRIVATE reset_protocol)
add_test(NAME protocol COMMAND reset_protocol_tests)
add_executable(reset_protocol_sim simulator.c)
target_link_libraries(reset_protocol_sim PRIVATE reset_protocol)
